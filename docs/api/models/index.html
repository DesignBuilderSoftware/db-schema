    class dsbXML(BaseModel):
    model_config = ConfigDict(populate_by_name=True)

    name: str = attr()
    date: datetime.date = attr()
    version: str = attr()
    objects: str = attr()
    site: Site | None = element(tag="Site", default=None)

    @classmethod
    def from_xml(cls, source: str | bytes) -> dsbXML:  # pylint: disable=arguments-differ
        from lxml import etree

        if isinstance(source, (str, bytes)):
            # huge_tree: large models can contain a single text node over
            # libxml2's default 10 MB limit, which it otherwise refuses with
            # "xmlSAX2Characters: huge text node". DesignBuilder's own files
            # are trusted input, so lifting the limit is acceptable here.
            root = etree.fromstring(source, parser=etree.XMLParser(huge_tree=True))
        else:
            root = source

        try:
            instance = super().from_xml_tree(root)
        except ValidationError as exc:
            raise ModelValidationError(exc, root=root) from exc
        return instance

    def to_xml(self, **kwargs) -> str | bytes:
        """The model as UTF-8 XML with a declaration: bytes, or str for encoding="unicode".

        The tree is written once; it used to be written, parsed back and written again only
        to set the root's tag.
        """
        from lxml import etree

        tree_args = ("skip_empty", "exclude_none", "exclude_unset")
        tree = {k: kwargs.pop(k) for k in tree_args if k in kwargs}
        as_text = kwargs.pop("encoding", None) in ("unicode", str)
        root = self.to_xml_tree(**tree)
        root.tag = "dsbXML"
        for el in root.iter():
            if el.text == "":  # written <X/>, as a parsed tree has it
                el.text = None
        kwargs.setdefault("xml_declaration", True)
        out = etree.tostring(root, encoding="UTF-8", **kwargs)
        return out.decode("UTF-8") if as_text else bytes(out)

    class AssemblyLibrary(BaseModel):
    assemblyHandle: int = attr()
    # DesignBuilder writes <AssemblyLibrary assemblyHandle="0"/> with no
    # <Assembly> children when a model has no assemblies.
    assembly: list[Assembly] = element(tag="Assembly", default_factory=list)

    @field_serializer("assemblyHandle")
    def serialize_assembly_handle(self, value: int) -> int:
        # The library's handle is the next free assembly handle, so it follows
        # the assemblies it holds. A library without assemblies keeps its own.
        if not self.assembly:
            return value
        return max(item.assemblyHandle for item in self.assembly) + 1
