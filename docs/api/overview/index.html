    def load_model(filepath: str | Path, use_cache: bool = True):
    path = Path(filepath) if isinstance(filepath, str) else filepath

    if path.suffix != ".xml":
        raise ValueError(
            f"Unsupported file format: {path.suffix}. Only .xml files are supported."
        )

    dsbXML = _get_dsbXML()
    if use_cache:
        dsb = _cached_load_model_internal(str(path.resolve()))
    else:
        with open(path, "rb") as f:
            dsb = dsbXML.from_xml(f.read())

    if dsb.site:
        set_site_counter(dsb.site.handle)
    return dsb

    def save_model(model, filepath: str | Path) -> None:
    path = Path(filepath) if isinstance(filepath, str) else filepath

    if path.suffix != ".xml":
        raise ValueError(
            f"Unsupported file format: {path.suffix}. Only .xml files are supported."
        )

    xml_output = model.to_xml()
    with open(path, "wb") as f:
        if isinstance(xml_output, bytes):
            f.write(xml_output)
        else:
            f.write(xml_output.encode("utf-8"))

    def validate_against_xsd(source) -> list[str]:
    """Check a model, an XML file path or XML bytes against DesignBuilder.xsd.

    Returns the schema's error messages; an empty list means the XML is valid.
    A loaded model is serialised first, so this also tells whether saving it
    would produce a file DesignBuilder's schema accepts.
    """
    from lxml import etree

    if isinstance(source, (str, Path)):
        document = etree.parse(str(source))
    else:
        xml = source if isinstance(source, bytes) else source.to_xml()
        xml = xml if isinstance(xml, bytes) else xml.encode("utf-8")
        document = etree.fromstring(xml).getroottree()
    schema = _xsd_schema()
    schema.validate(document)
    # lxml's error_log is a C extension type pylint cannot see is iterable.
    return [error.message for error in schema.error_log]  # pylint: disable=not-an-iterable

    Comparing models¶

    Semantic diff between two loaded models.

    Reports added/removed buildings, zones, surfaces, and openings, changed attributes (old -> new), and HVAC topology changes. Handle/index fields on ObjectIDs (handle, buildingHandle, buildingBlockHandle, zoneHandle, surfaceIndex, openingIndex) are ignored throughout — DesignBuilder renumbers these on every save, so they're noise, not a semantic change.

    @dataclass
class ModelDiff:
    added: dict[str, list] = field(default_factory=dict)
    removed: dict[str, list] = field(default_factory=dict)
    attribute_changes: list[dict] = field(default_factory=list)

    @property
    def has_changes(self) -> bool:
        return bool(self.added or self.removed or self.attribute_changes)

    def summary(self) -> str:
        lines = []
        for label, items in self.added.items():
            lines.append(f"+ {len(items)} {label} added")
        for label, items in self.removed.items():
            lines.append(f"- {len(items)} {label} removed")
        for change in self.attribute_changes:
            lines.append(f"~ {change['path']}: {change['old']!r} -> {change['new']!r}")
        return "\n".join(lines) if lines else "No semantic changes"

    Errors¶

    Validation error formatting with model context (building/zone/surface).

    Pydantic's default ValidationError shows the field path as raw indices (site.buildings.building.0...zones.zone.3...surfaces.surface.2.area), which doesn't say which building or zone that is without cross-referencing the source file by hand. ModelValidationError re-walks the original XML tree along that same path and adds the building title / zone handle / surface index actually found there.

    class ModelValidationError(ValueError):
    """Wraps a pydantic ValidationError, adding model context to each error.

    Falls back to pydantic's own field path when the source XML tree isn't
    available, or an error's path leaves the mapped building/zone/surface
    hierarchy (e.g. HVAC fields).
    """

    def __init__(self, original: ValidationError, root=None):
        self.original = original
        self.errors = original.errors()
        super().__init__(self._format(root))

    def _format(self, root) -> str:
        lines = [f"{len(self.errors)} validation error(s) for {self.original.title}:"]
        for error in self.errors:
            loc = error.get("loc", ())
            crumb = _breadcrumb(loc, root)
            field = loc[-1] if loc else "?"
            path = ".".join(str(part) for part in loc)
            if crumb:
                lines.append(f"  {crumb} > {field}: {error['msg']} ({path})")
            else:
                lines.append(f"  {path}: {error['msg']}")
        return "\n".join(lines)

    class Point3D(BaseModel):
    value: str = "0;0;0"

    @field_validator("value", mode="before")
    @classmethod
    def _coerce_empty(cls, v):
        """Treat empty or whitespace-only text as the origin sentinel."""
        if not v or not str(v).strip():
            return "0;0;0"
        return v

    def _parse_coords(self) -> tuple[float, float, float]:
        parsed = tuple(float(c) for c in self.value.split(";"))
        if len(parsed) != 3:
            raise ValueError(f"Expected 3 coordinates, got {len(parsed)}: {self.value!r}")
        return parsed

    @property
    def x(self) -> float:
        return self._parse_coords()[0]

    @property
    def y(self) -> float:
        return self._parse_coords()[1]

    @property
    def z(self) -> float:
        return self._parse_coords()[2]

    @property
    def coords(self) -> list[float]:
        return list(self._parse_coords())

    @property
    def is_zero(self) -> bool:
        """True when all coordinates are zero — the sentinel for 'no position'."""
        return self.value == "0;0;0" or all(c == 0.0 for c in self._parse_coords())

    def __setattr__(self, name: str, value: float | list[float]) -> None:
        if name == "coords":
            if isinstance(value, list) and len(value) == 3:
                self.value = ";".join(map(str, value))
            else:
                raise ValueError("Coordinates must be a list of 3 numbers")
        elif name in ["x", "y", "z"]:
            index = {"x": 0, "y": 1, "z": 2}[name]
            coords = self.value.split(";")
            coords[index] = str(value)
            self.value = ";".join(coords)
        else:
            super().__setattr__(name, value)

    value = '0;0;0' class-attribute instance-attribute ¶

    WrappedPoint3D ¶

    Bases: BaseModel

    A Point3D nested one level deeper under its own child tag.

    Most single-coordinate fields in the DesignBuilder XML (Origin, Coordinate, Begin, End, ControlPoint, ...) are not the coordinate text themselves but a wrapper element containing a single child, e.g. 1;2;3. Typing these fields as plain Point3D silently parses to the "0;0;0" default, since pydantic-xml looks for the coordinate text directly on the wrapper element rather than recursing into the nested tag. Exposes the same x/y/z/coords/ is_zero interface as Point3D so existing call sites are unaffected.

    class WrappedPoint3D(BaseModel):
    """A Point3D nested one level deeper under its own <Point3D> child tag.

    Most single-coordinate fields in the DesignBuilder XML (Origin,
    Coordinate, Begin, End, ControlPoint, ...) are not the coordinate text
    themselves but a wrapper element containing a single <Point3D> child,
    e.g. <Origin><Point3D>1;2;3</Point3D></Origin>. Typing these fields as
    plain Point3D silently parses to the "0;0;0" default, since pydantic-xml
    looks for the coordinate text directly on the wrapper element rather
    than recursing into the nested tag. Exposes the same x/y/z/coords/
    is_zero interface as Point3D so existing call sites are unaffected.
    """

    point_3d: Point3D = element(tag="Point3D", default_factory=Point3D)

    @property
    def x(self) -> float:
        return self.point_3d.x

    @property
    def y(self) -> float:
        return self.point_3d.y

    @property
    def z(self) -> float:
        return self.point_3d.z

    @property
    def coords(self) -> list[float]:
        return self.point_3d.coords

    @property
    def is_zero(self) -> bool:
        return self.point_3d.is_zero

    class BaseModel(BaseXmlModel, search_mode=SearchMode.UNORDERED):
    """Base XML model with unordered element search.

    UNORDERED: pydantic-xml's default ORDERED mode consumes elements
    sequentially, so any field that scans past a later-appearing tag makes
    it invisible to subsequent fields. UNORDERED finds every element by tag
    name regardless of position — required because DesignBuilder XML places
    the same tags in different orders across component types.
    """

    model_config = ConfigDict(
        populate_by_name=True,
        validate_assignment=True,
        str_strip_whitespace=True,
        slots=True,
    )

    @classmethod
    def _built(cls) -> None:
        """Build the class's schema now if import left it for first use (see __init__.py)."""
        if cls.__xml_serializer__ is None:
            import dsbxml

            dsbxml._load_core_module()  # pylint: disable=protected-access
            cls.model_rebuild()

    @classmethod
    def from_xml_tree(cls, root, *args, **kwargs):
        cls._built()
        return super().from_xml_tree(root, *args, **kwargs)

    def to_xml_tree(self, **kwargs):
        """Serialise without unset optional values.

        pydantic-xml writes an unset optional attribute as ``name=""`` and an
        unset optional primitive element as an empty tag, both invalid
        against the schema. Empty strings and empty containers are not None
        and are still written.
        """
        kwargs.setdefault("exclude_none", True)
        type(self)._built()
        return super().to_xml_tree(**kwargs)

    def to_xml(self, **kwargs):
        """Serialise to XML text; see to_xml_tree for the unset-value handling."""
        kwargs.setdefault("exclude_none", True)
        return super().to_xml(**kwargs)

    model_config = ConfigDict(populate_by_name=True, validate_assignment=True, str_strip_whitespace=True, slots=True) class-attribute instance-attribute ¶
